🔧 명령줄 다운로드

모든 파일을 한번에 다운로드하려면:

# 메인 온톨로지
curl -O https://5173-iq167hrnwb18xvsq7sd61-de59bda9.sandbox.novita.ai/coffeeland.rdf

# generated_imports 폴더 생성
mkdir -p generated_imports

# 모든 generated_imports 파일
cd generated_imports
for file in certification consumer cooperative farmer price processingmethod processingmill qualitygrade retailer roaster season tradeagreement; do
  curl -O https://5173-iq167hrnwb18xvsq7sd61-de59bda9.sandbox.novita.ai/generated_imports/${file}_individuals.rdf
done
