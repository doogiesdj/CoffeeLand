  <rdf:type rdf:resource="...#Farm"/>
  
  <!-- Data Properties -->
  <hasName>Yirgacheffe_Kochere_Farm</hasName>
  <hasRegion>Yirgacheffe</hasRegion>
  <hasAltitudeRange>1800-2200</hasAltitudeRange>
  <hasFarmSize>Medium</hasFarmSize>
  <hasAnnualProduction rdf:datatype="xsd:integer">150</hasAnnualProduction>
  
  <!-- Object Properties -->
  <isLocatedIn rdf:resource="...#Ethiopia"/>
  <cultivates rdf:resource="...#Arabica_Heirloom"/>
  <usesProcessingMethod rdf:resource="...#Washed"/>
  <memberOf rdf:resource="...#Cooperative_Yirgacheffe"/>
  
  <!-- Multiple Certifications -->
  <hasCertification rdf:resource="...#Fair_Trade"/>
  <hasCertification rdf:resource="...#Organic"/>
</owl:NamedIndividual>
